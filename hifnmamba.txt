# %% --- Cell 5: Mamba (S6) block — pure PyTorch ---
class MambaBlock(nn.Module):
    """
    Selective State Space (S6) block — Gu & Dao (2023), pure PyTorch.

    Forward:
        x: (B, L, d_model) -> (B, L, d_model)
    Sequential scan in PyTorch is fine for small L (here L=16);
    swap to mamba-ssm's selective_scan_fn if you want kernel speed on long seqs.
    """
    def __init__(self, d_model, d_state=16, d_conv=4, expand=2):
        super().__init__()
        self.d_model = d_model
        self.d_state = d_state
        self.d_conv  = d_conv
        self.d_inner = expand * d_model

        self.in_proj = nn.Linear(d_model, 2 * self.d_inner, bias=False)
        self.conv1d  = nn.Conv1d(self.d_inner, self.d_inner,
                                 kernel_size=d_conv, groups=self.d_inner,
                                 padding=d_conv - 1, bias=True)
        # x -> (Δ, B_param, C_param)
        self.x_proj  = nn.Linear(self.d_inner,
                                 self.d_inner + 2 * d_state, bias=False)

        # A parametrized as -exp(A_log) for stability (negative real eigenvalues)
        A = torch.arange(1, d_state + 1, dtype=torch.float32).repeat(self.d_inner, 1)
        self.A_log = nn.Parameter(torch.log(A))
        self.D     = nn.Parameter(torch.ones(self.d_inner))   # skip
        self.out_proj = nn.Linear(self.d_inner, d_model, bias=False)

    def forward(self, x):
        B, L, _ = x.shape
        xz = self.in_proj(x)                                  # (B, L, 2*d_inner)
        x_in, z = xz.chunk(2, dim=-1)                         # each (B, L, d_inner)

        # 1D causal-ish conv along sequence
        x_in = x_in.transpose(1, 2)
        x_in = self.conv1d(x_in)[:, :, :L]
        x_in = x_in.transpose(1, 2)
        x_in = F.silu(x_in)

        # selective parameters from input
        proj = self.x_proj(x_in)
        delta, B_par, C_par = torch.split(
            proj, [self.d_inner, self.d_state, self.d_state], dim=-1
        )
        delta = F.softplus(delta)                             # (B, L, d_inner)

        A = -torch.exp(self.A_log)                            # (d_inner, d_state)

        # discretize
        dA = torch.exp(delta.unsqueeze(-1) * A.unsqueeze(0).unsqueeze(0))     # (B,L,di,ds)
        dB = delta.unsqueeze(-1) * B_par.unsqueeze(-2)                        # (B,L,di,ds)
        dBx = dB * x_in.unsqueeze(-1)                                         # (B,L,di,ds)

        # sequential selective scan (L is small here)
        h = torch.zeros(B, self.d_inner, self.d_state, device=x.device, dtype=x.dtype)
        ys = []
        for t in range(L):
            h = dA[:, t] * h + dBx[:, t]
            ys.append((h * C_par[:, t].unsqueeze(1)).sum(dim=-1))
        y = torch.stack(ys, dim=1)                            # (B, L, d_inner)

        y = y + self.D * x_in                                 # skip
        y = y * F.silu(z)                                     # gate
        return self.out_proj(y)


# %% --- Cell 6: HIFN layer (4 learnable info-theoretic parameters) ---
class HIFNLayer(nn.Module):
    """
    Hybrid Information Flow Network layer.

    Four learnable parameters per layer:
        β  (log_beta)         : information retention / KL weight
        B  (log_entropy_budget): entropy capacity cap H(z) <= B
        c  (logit_compression) : magnitude scaling in [0.3, 1.0]
        g  (gates_logit)       : static, global per-feature gate (interpretable)
    """
    def __init__(self, d_in: int, d_out: int):
        super().__init__()
        self.d_in, self.d_out = d_in, d_out
        self.encoder    = nn.Sequential(
            nn.Linear(d_in, d_out * 2),
            nn.LayerNorm(d_out * 2),
            nn.GELU(),
        )
        self.to_mu      = nn.Linear(d_out * 2, d_out)
        self.to_logvar  = nn.Linear(d_out * 2, d_out)

        # learnable info-theoretic parameters
        self.log_beta            = nn.Parameter(torch.tensor(-2.0))
        self.log_entropy_budget  = nn.Parameter(torch.tensor(float(np.log(d_out))))
        self.logit_compression   = nn.Parameter(torch.tensor(1.0))
        self.gates_logit         = nn.Parameter(torch.zeros(d_in))

    def forward(self, x, training: bool = True):
        gates = torch.sigmoid(self.gates_logit)               # (d_in,)
        x_g   = x * gates

        h     = self.encoder(x_g)
        mu    = self.to_mu(h)
        logvar = self.to_logvar(h).clamp(-10.0, 10.0)

        c     = 0.3 + 0.7 * torch.sigmoid(self.logit_compression)
        mu    = mu * c

        if training:
            z = mu + torch.randn_like(mu) * torch.exp(0.5 * logvar)
        else:
            z = mu

        beta  = torch.exp(self.log_beta)
        kl    = -0.5 * (1 + logvar - mu.pow(2) - logvar.exp()).sum(-1).mean()
        kl    = beta * kl

        H_z   = 0.5 * (1 + math.log(2 * math.pi) + logvar).sum(-1).mean()
        B_tgt = torch.exp(self.log_entropy_budget)
        ent_pen = F.relu(H_z - B_tgt)

        return z, kl, ent_pen, gates


# %% --- Cell 7: HIFN-Mamba model ---
class HIFNMamba(nn.Module):
    """
    HIFN trunk -> tokenize to (seq_len, d_model) -> Mamba stack -> classifier.
    Static input gates remain the interpretable handle (same as HIFN-T).
    """
    def __init__(self, n_features: int, n_classes: int, cfg=CFG):
        super().__init__()
        self.seq_len = cfg["seq_len"]
        self.d_model = cfg["d_model"]

        self.hifn1 = HIFNLayer(n_features, cfg["d_hidden_1"])
        self.hifn2 = HIFNLayer(cfg["d_hidden_1"], cfg["seq_len"] * cfg["d_model"])

        self.pos_embed = nn.Parameter(torch.zeros(1, cfg["seq_len"], cfg["d_model"]))
        nn.init.trunc_normal_(self.pos_embed, std=0.02)

        self.norms  = nn.ModuleList([nn.LayerNorm(cfg["d_model"]) for _ in range(cfg["n_mamba"])])
        self.mambas = nn.ModuleList([
            MambaBlock(cfg["d_model"], d_state=cfg["d_state"],
                       d_conv=cfg["d_conv"], expand=cfg["expand"])
            for _ in range(cfg["n_mamba"])
        ])
        self.head_norm   = nn.LayerNorm(cfg["d_model"])
        self.classifier  = nn.Linear(cfg["d_model"], n_classes)

    def forward(self, x, training: bool = True):
        z1, kl1, ent1, g1 = self.hifn1(x,  training)
        z2, kl2, ent2, _  = self.hifn2(z1, training)

        tokens = z2.view(x.size(0), self.seq_len, self.d_model) + self.pos_embed
        for norm, mamba in zip(self.norms, self.mambas):
            tokens = tokens + mamba(norm(tokens))             # pre-norm residual
        feat   = self.head_norm(tokens.mean(dim=1))
        logits = self.classifier(feat)
        return logits, kl1 + kl2, ent1 + ent2, g1             # g1 = input-feature gates


# %% --- Cell 8: Train ---
model = HIFNMamba(N_FEAT, N_CLS).to(DEVICE)
n_params = sum(p.numel() for p in model.parameters())
print(f"HIFN-Mamba: {n_params/1e6:.2f}M params")

# class weights (CICIDS Friday is fairly balanced post-clean, but include for safety)
counts = np.bincount(y_tr)
cls_w  = torch.tensor(counts.sum() / (N_CLS * counts), dtype=torch.float32, device=DEVICE)

opt   = torch.optim.AdamW(model.parameters(), lr=CFG["lr"], weight_decay=CFG["weight_decay"])
sched = torch.optim.lr_scheduler.CosineAnnealingLR(opt, T_max=CFG["epochs"])
ce    = nn.CrossEntropyLoss(weight=cls_w)

hist = {k: [] for k in ["tr_loss", "tr_acc", "te_acc", "te_f1", "kl", "ent", "lr"]}

for ep in range(1, CFG["epochs"] + 1):
    model.train()
    t0 = time.time()
    n_seen = correct = 0; tot_loss = tot_kl = tot_ent = 0.0
    for xb, yb in train_ld:
        xb, yb = xb.to(DEVICE, non_blocking=True), yb.to(DEVICE, non_blocking=True)
        opt.zero_grad()
        logits, kl, ent, gates = model(xb, training=True)
        loss = (ce(logits, yb)
                + CFG["lambda_kl"]   * kl
                + CFG["lambda_ent"]  * ent
                + CFG["lambda_gate"] * gates.abs().sum())
        loss.backward()
        torch.nn.utils.clip_grad_norm_(model.parameters(), 1.0)
        opt.step()
        n_seen  += yb.size(0)
        correct += (logits.argmax(-1) == yb).sum().item()
        tot_loss += loss.item() * yb.size(0)
        tot_kl   += kl.item()   * yb.size(0)
        tot_ent  += ent.item()  * yb.size(0)
    sched.step()

    # eval
    model.eval()
    yp_all, yt_all = [], []
    with torch.no_grad():
        for xb, yb in test_ld:
            xb = xb.to(DEVICE, non_blocking=True)
            logits, _, _, _ = model(xb, training=False)
            yp_all.append(logits.argmax(-1).cpu().numpy())
            yt_all.append(yb.numpy())
    yp = np.concatenate(yp_all); yt = np.concatenate(yt_all)
    te_acc = accuracy_score(yt, yp); te_f1 = f1_score(yt, yp, average="macro")

    hist["tr_loss"].append(tot_loss / n_seen)
    hist["tr_acc"] .append(correct / n_seen)
    hist["te_acc"] .append(te_acc)
    hist["te_f1"]  .append(te_f1)
    hist["kl"]     .append(tot_kl / n_seen)
    hist["ent"]    .append(tot_ent / n_seen)
    hist["lr"]     .append(opt.param_groups[0]["lr"])

    print(f"[{ep:02d}/{CFG['epochs']}]  loss {hist['tr_loss'][-1]:.4f}  "
          f"tr_acc {hist['tr_acc'][-1]*100:.2f}%  "
          f"te_acc {te_acc*100:.2f}%  te_f1 {te_f1:.4f}  "
          f"kl {hist['kl'][-1]:.2f}  ent {hist['ent'][-1]:.2f}  "
          f"({time.time()-t0:.1f}s)")

# %% --- Cell 10: Mamba (S6) block — pure PyTorch ---
class MambaBlock(nn.Module):
    """Selective State Space (S6) block — Gu & Dao 2023, pure-PyTorch sequential scan."""
    def __init__(self, d_model, d_state=16, d_conv=4, expand=2):
        super().__init__()
        self.d_model = d_model
        self.d_state = d_state
        self.d_inner = expand * d_model

        self.in_proj  = nn.Linear(d_model, 2 * self.d_inner, bias=False)
        self.conv1d   = nn.Conv1d(self.d_inner, self.d_inner,
                                  kernel_size=d_conv, groups=self.d_inner,
                                  padding=d_conv - 1, bias=True)
        self.x_proj   = nn.Linear(self.d_inner, self.d_inner + 2 * d_state, bias=False)
        A = torch.arange(1, d_state + 1, dtype=torch.float32).repeat(self.d_inner, 1)
        self.A_log    = nn.Parameter(torch.log(A))
        self.D        = nn.Parameter(torch.ones(self.d_inner))
        self.out_proj = nn.Linear(self.d_inner, d_model, bias=False)

    def forward(self, x):
        B, L, _ = x.shape
        xz = self.in_proj(x); x_in, z = xz.chunk(2, dim=-1)
        x_in = self.conv1d(x_in.transpose(1, 2))[:, :, :L].transpose(1, 2)
        x_in = F.silu(x_in)
        proj = self.x_proj(x_in)
        delta, B_par, C_par = torch.split(
            proj, [self.d_inner, self.d_state, self.d_state], dim=-1)
        delta = F.softplus(delta)
        A = -torch.exp(self.A_log)
        dA  = torch.exp(delta.unsqueeze(-1) * A.unsqueeze(0).unsqueeze(0))
        dBx = (delta.unsqueeze(-1) * B_par.unsqueeze(-2)) * x_in.unsqueeze(-1)
        h = torch.zeros(B, self.d_inner, self.d_state, device=x.device, dtype=x.dtype)
        ys = []
        for t in range(L):
            h = dA[:, t] * h + dBx[:, t]
            ys.append((h * C_par[:, t].unsqueeze(1)).sum(-1))
        y = torch.stack(ys, dim=1) + self.D * x_in
        return self.out_proj(y * F.silu(z))


# %% --- Cell 11: HIFN layer (β, B, c, gates) ---
class HIFNLayer(nn.Module):
    def __init__(self, d_in, d_out):
        super().__init__()
        self.encoder = nn.Sequential(
            nn.Linear(d_in, d_out * 2),
            nn.LayerNorm(d_out * 2),
            nn.GELU(),
        )
        self.to_mu     = nn.Linear(d_out * 2, d_out)
        self.to_logvar = nn.Linear(d_out * 2, d_out)
        self.log_beta            = nn.Parameter(torch.tensor(-2.0))
        self.log_entropy_budget  = nn.Parameter(torch.tensor(float(np.log(d_out))))
        self.logit_compression   = nn.Parameter(torch.tensor(1.0))
        self.gates_logit         = nn.Parameter(torch.zeros(d_in))

    def forward(self, x, training=True):
        gates = torch.sigmoid(self.gates_logit)
        h     = self.encoder(x * gates)
        mu    = self.to_mu(h)
        lv    = self.to_logvar(h).clamp(-10, 10)
        c     = 0.3 + 0.7 * torch.sigmoid(self.logit_compression)
        mu    = mu * c
        z     = mu + torch.randn_like(mu) * torch.exp(0.5 * lv) if training else mu
        kl    = torch.exp(self.log_beta) * (-0.5 * (1 + lv - mu.pow(2) - lv.exp()).sum(-1).mean())
        H_z   = 0.5 * (1 + math.log(2 * math.pi) + lv).sum(-1).mean()
        ent   = F.relu(H_z - torch.exp(self.log_entropy_budget))
        return z, kl, ent, gates


# %% --- Cell 12: HIFN-Mamba ---
class HIFNMamba(nn.Module):
    def __init__(self, n_features, n_classes, cfg=CFG):
        super().__init__()
        self.seq_len = cfg["seq_len"]; self.d_model = cfg["d_model"]
        self.hifn1   = HIFNLayer(n_features, cfg["d_hidden_1"])
        self.hifn2   = HIFNLayer(cfg["d_hidden_1"], cfg["seq_len"] * cfg["d_model"])
        self.pos     = nn.Parameter(torch.zeros(1, cfg["seq_len"], cfg["d_model"]))
        nn.init.trunc_normal_(self.pos, std=0.02)
        self.norms   = nn.ModuleList([nn.LayerNorm(cfg["d_model"]) for _ in range(cfg["n_mamba"])])
        self.mambas  = nn.ModuleList([
            MambaBlock(cfg["d_model"], d_state=cfg["d_state"],
                       d_conv=cfg["d_conv"], expand=cfg["expand"])
            for _ in range(cfg["n_mamba"])])
        self.head_norm  = nn.LayerNorm(cfg["d_model"])
        self.classifier = nn.Linear(cfg["d_model"], n_classes)

    def forward(self, x, training=True, return_embed=False):
        z1, kl1, ent1, g1 = self.hifn1(x,  training)
        z2, kl2, ent2, _  = self.hifn2(z1, training)
        tokens = z2.view(x.size(0), self.seq_len, self.d_model) + self.pos
        for norm, mb in zip(self.norms, self.mambas):
            tokens = tokens + mb(norm(tokens))
        feat   = self.head_norm(tokens.mean(dim=1))
        logits = self.classifier(feat)
        if return_embed:
            return logits, kl1 + kl2, ent1 + ent2, g1, feat
        return logits, kl1 + kl2, ent1 + ent2, g1
